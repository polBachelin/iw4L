#using_animtree( "generic_human" );

setting( key )
{
	return tablelookup( "survival/settings.csv", 0, key, 1 );
}

settingseconds( key )
{
	return int( setting( key ) ) / 1000;
}

settingpercent( key )
{
	return int( setting( key ) ) / 100;
}

// survival/actor_anims.csv: state, stance ("" for any), direction ("" for
// any), anim names (space separated, one drawn per play). The engine
// narrows each cell to the loaded anims before the script starts; a cell
// left empty plays nothing.
readanims()
{
	if ( isdefined( level.actoranims ) )
		return;
	table = "survival/actor_anims.csv";
	level.actoranims = [];
	for ( row = 0; tablelookupbyrow( table, row, 0 ) != ""; row++ )
	{
		state = tablelookupbyrow( table, row, 0 );
		if ( !isdefined( level.actoranims[ state ] ) )
			level.actoranims[ state ] = [];
		entry = spawnstruct();
		entry.stance = tablelookupbyrow( table, row, 1 );
		entry.direction = tablelookupbyrow( table, row, 2 );
		entry.anims = strtok( tablelookupbyrow( table, row, 3 ), " " );
		level.actoranims[ state ][ level.actoranims[ state ].size ] = entry;
	}
}

// One anim of `state` for the actor's stance and `direction`, from the first
// row that fits, else the state's first row; "" when there is none.
pickanim( state, direction )
{
	rows = level.actoranims[ state ];
	if ( !isdefined( rows ) || rows.size == 0 )
		return "";
	picked = rows[ 0 ];
	for ( i = 0; i < rows.size; i++ )
	{
		if ( rows[ i ].stance != "" && rows[ i ].stance != self.pose )
			continue;
		if ( rows[ i ].direction != "" && rows[ i ].direction != direction )
			continue;
		picked = rows[ i ];
		break;
	}
	if ( picked.anims.size == 0 )
		return "";
	return picked.anims[ randomint( picked.anims.size ) ];
}

// Loops an anim of `state`, unless one of it already plays.
playloop( state )
{
	if ( isdefined( self.loopstate ) && self.loopstate == state && self.loopstance == self.pose )
		return;
	name = self pickanim( state, "" );
	self.loopstate = state;
	self.loopstance = self.pose;
	if ( name != "" )
		self setanimknob( name, 1, settingseconds( "actor_blend_ms" ) );
}

// Plays an anim of `state` once and returns when it ends.
playonce( state, direction )
{
	self.loopstate = undefined;
	name = self pickanim( state, direction );
	if ( name == "" )
		return;
	self setflaggedanimknob( "actor_anim", name, 1, settingseconds( "actor_blend_ms" ) );
	for ( ;; )
	{
		self waittill( "actor_anim", note );
		if ( note == "end" )
			return;
	}
}

// Entry point for an actor of `archetype` spawned by spawnactor and dressed:
// it runs to the nearest living human, engages the one it sees in range,
// flinches when hurt and plays its death. Accuracy (c5), burst length (c6)
// and speed (c4) are its survival/archetypes.csv row's.
actor_think( archetype )
{
	readanims();
	self useanimtree( #animtree );
	table = "survival/archetypes.csv";
	self.archetype = archetype;
	self.pose = "stand";
	self.aimaccuracy = int( tablelookup( table, 0, archetype, 5 ) ) / 100;
	self.burstshots = int( tablelookup( table, 0, archetype, 6 ) );
	self.runrate = 1;
	speed = tablelookup( table, 0, archetype, 4 );
	if ( speed != "" )
		self.runrate = int( speed ) / 100;
	self.gun = self actorweapon( archetype );
	if ( self.gun != "" )
		self setweapon( self.gun );
	self setgoalradius( int( setting( "actor_goal_radius" ) ) );
	self.hitdirection = "b";
	self.inpain = false;
	self thread actorpain();
	self thread actordeath();
	self actorloop();
}

actorweapon( archetype )
{
	weapon = tablelookup( "survival/archetypes.csv", 0, archetype, 1 );
	if ( weapon == "" )
		weapon = tablelookup( setting( "waves_table" ), 1, archetype, 5 );
	alias = tablelookup( "survival/weapon_aliases.csv", 0, weapon, 1 );
	if ( alias != "" )
		return alias;
	return weapon;
}

actorloop()
{
	self endon( "death" );
	for ( ;; )
	{
		while ( self.inpain )
			wait 0.05;
		target = self nearesthuman( true );
		if ( isdefined( target ) && distance( self.origin, target.origin ) <= int( setting( "actor_engage_range" ) ) )
		{
			self engage( target );
			continue;
		}
		if ( !isdefined( target ) )
			target = self nearesthuman( false );
		if ( isdefined( target ) )
			self movetoward( target );
		else
			self idle();
		wait settingseconds( "actor_think_ms" );
	}
}

idle()
{
	self stopmoving();
	self playloop( "idle" );
}

stopmoving()
{
	if ( isdefined( self.moving ) )
		self setgoalpos( self.origin );
	self.moving = undefined;
}

// Runs toward `target`, walking once within actor_walk_range.
movetoward( target )
{
	self.pose = "stand";
	state = "run";
	rate = self.runrate;
	if ( distance( self.origin, target.origin ) <= int( setting( "actor_walk_range" ) ) )
	{
		state = "walk";
		rate = rate * settingpercent( "actor_walk_rate_percent" );
	}
	self.moveplaybackrate = rate;
	self setgoalpos( target.origin );
	self.moving = true;
	self playloop( state );
}

// Stops, picks a stance, aims and fires one burst at `target`, then rests.
engage( target )
{
	self stopmoving();
	self.pose = "stand";
	if ( randomint( 100 ) < int( setting( "actor_crouch_percent" ) ) )
		self.pose = "crouch";
	self faceat( target );
	self playloop( "aim" );
	wait settingseconds( "actor_blend_ms" );
	if ( self.inpain )
		return;
	if ( self.gun == "" || self.burstshots <= 0 )
	{
		wait settingseconds( "actor_think_ms" );
		return;
	}
	interval = weaponfiretime( self.gun );
	if ( interval <= 0 )
		interval = 0.05;
	self playloop( "fire" );
	for ( shot = 0; shot < self.burstshots && !self.inpain; shot++ )
	{
		if ( !isalive( target ) || !self cansee( target ) )
			break;
		self faceat( target );
		self shoot( self aimpoint( target ) );
		wait interval;
	}
	if ( !self.inpain )
		self playloop( "aim" );
	wait settingseconds( "actor_burst_rest_ms" );
}

faceat( target )
{
	angles = vectortoangles( target.origin - self.origin );
	self.angles = ( 0, angles[ 1 ], 0 );
}

// Where a shot goes: the target's eye at the archetype's accuracy, else a
// point up to actor_miss_spread units off it.
aimpoint( target )
{
	point = target geteye();
	if ( randomfloat( 1 ) < self.aimaccuracy )
		return point;
	spread = int( setting( "actor_miss_spread" ) );
	return point + ( randomfloatrange( 0 - spread, spread ), randomfloatrange( 0 - spread, spread ), randomfloatrange( 0 - spread, spread ) );
}

eye()
{
	if ( self.pose == "crouch" )
		return self.origin + ( 0, 0, int( setting( "actor_crouch_eye_height" ) ) );
	return self.origin + ( 0, 0, int( setting( "actor_eye_height" ) ) );
}

cansee( target )
{
	return bullettracepassed( self eye(), target geteye(), false, self );
}

// The nearest living human of the other side, only one the actor sees when
// `visible`.
nearesthuman( visible )
{
	if ( self getteam() == setting( "human_team" ) || !isdefined( level.humans ) )
		return undefined;
	nearest = undefined;
	best = 0;
	for ( i = 0; i < level.humans.size; i++ )
	{
		human = level.humans[ i ];
		if ( !isalive( human ) )
			continue;
		dist = distancesquared( self.origin, human.origin );
		if ( isdefined( nearest ) && dist >= best )
			continue;
		if ( visible && !self cansee( human ) )
			continue;
		nearest = human;
		best = dist;
	}
	return nearest;
}

// The way a hit along `direction` pushes the body, against the actor's
// facing: "f" forward (hit from behind), "b" back, "l" left, "r" right.
pushdirection( direction )
{
	forward = anglestoforward( self.angles );
	along = vectordot( direction, forward );
	if ( along > 0.707 )
		return "f";
	if ( along < -0.707 )
		return "b";
	if ( vectordot( direction, anglestoright( self.angles ) ) > 0 )
		return "r";
	return "l";
}

actorpain()
{
	self endon( "death" );
	for ( ;; )
	{
		self waittill( "damage", amount, attacker, direction );
		if ( isdefined( direction ) && length( direction ) > 0 )
			self.hitdirection = self pushdirection( direction );
		if ( self.inpain || !self isalive() || randomint( 100 ) >= int( setting( "actor_pain_percent" ) ) )
			continue;
		self.inpain = true;
		self stopmoving();
		self playonce( "pain", self.hitdirection );
		self.inpain = false;
	}
}

isexplosive( means )
{
	kinds = strtok( setting( "actor_explosive_means" ), " " );
	for ( i = 0; i < kinds.size; i++ )
	{
		if ( kinds[ i ] == means )
			return true;
	}
	return false;
}

// Plays a death of the actor's stance away from the killing hit, an
// explosive one for explosive means, and leaves the body for
// actor_corpse_ms.
actordeath()
{
	self waittill( "death", attacker, means );
	self.moving = undefined;
	state = "death";
	if ( isdefined( means ) && isexplosive( means ) )
		state = "explode";
	self playonce( state, self.hitdirection );
	wait settingseconds( "actor_corpse_ms" );
	self delete();
}
